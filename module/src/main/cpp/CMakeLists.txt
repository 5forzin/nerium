project(devicelab LANGUAGES CXX ASM)
if(ANDROID_ABI STREQUAL "arm64-v8a")
    add_library(guestprofile SHARED guest_profile.cpp)
    target_compile_features(guestprofile PRIVATE cxx_std_17)
    target_compile_options(guestprofile PRIVATE -Wall -Wextra -Werror -fvisibility=hidden)
    target_link_libraries(guestprofile PRIVATE dl)
    target_link_options(guestprofile PRIVATE -Wl,-z,max-page-size=16384)
    return()
endif()
